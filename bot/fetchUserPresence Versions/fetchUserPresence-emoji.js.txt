const activityTypeMap = {
  0: "Playing",
  1: "Streaming",
  2: "Listening to",
  3: "Watching",
  4: "Status:", // custom
  5: "Competing in",
};

// Helper to convert unicode emoji to CDN URL (cdnjs Twemoji)
function getTwemojiUrl(unicodeEmoji) {
  const codePoints = Array.from(unicodeEmoji).map(char => char.codePointAt(0).toString(16));
  return `https://cdnjs.cloudflare.com/ajax/libs/twemoji/14.0.2/72x72/${codePoints.join('-')}.png`;
}

// Function to fetch and return a specific user's presence data
async function fetchUserPresence(client, userId) {
  try {
    const guild = client.guilds.cache.first(); // Get the first available guild
    if (!guild) {
      return {
        status: "",
        activities: [],
        error: "No guilds available for the bot.",
      };
    }

    const member = await guild.members.fetch(userId);
    if (!member) {
      return {
        status: "",
        activities: [],
        error: "User not found in the guild.",
      };
    }

    const presence = member.presence || null;

    // Default response structure
    const response = {
      status: "offline",
      activities: [],
    };

    if (presence) {
      // Set online status
      response.status = presence.status || "offline";

      // Loop through all activities
      for (const activity of presence.activities) {
        let emojiUrl = null;

        if (activity.type === 4 && activity.emoji) {
          if (activity.emoji.id) {
            // Custom emoji
            const format = activity.emoji.animated ? "gif" : "png";
            emojiUrl = `https://cdn.discordapp.com/emojis/${activity.emoji.id}.${format}`;
          } else if (activity.emoji.name) {
            // Unicode emoji
            emojiUrl = getTwemojiUrl(activity.emoji.name);
          }
        }

        response.activities.push({
          activityType: activityTypeMap[activity.type] || "",
          activityText: activity.state || "",
          activityDetails: activity.details || "",
          activityImage: activity.assets?.largeImageURL() || "",
          activityTimestamps: activity.timestamps
            ? {
                start: activity.timestamps.start || null,
                end: activity.timestamps.end || null,
              }
            : null,
          activityEmoji: emojiUrl,
        });
      }
    } else {
      response.status = "invisible";
    }

    return response;
  } catch (error) {
    console.error("Error fetching user presence:", error);
    return {
      status: "",
      activities: [],
      error: "Failed to retrieve presence data.",
    };
  }
}

module.exports = fetchUserPresence;
